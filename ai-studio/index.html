const source = ctx.createBufferSource(); source.buffer = inputBuffer; /* 1️⃣ Analog Warmth */ const saturator = ctx.createWaveShaper(); const curve = new Float32Array(2048); for (let i = 0; i < 2048; i++) { const x = (i / 2048) * 2 - 1; curve[i] = Math.tanh(2.8 * x); } saturator.curve = curve; /* 2️⃣ Glue Compressor (soul glue) */ const comp = ctx.createDynamicsCompressor(); comp.threshold.value = -18; comp.knee.value = 12; comp.ratio.value = 2.8; comp.attack.value = 0.003; comp.release.value = 0.22; /* 3️⃣ EQ — Earth/Breath Balance */ const low = ctx.createBiquadFilter(); low.type = "lowshelf"; low.frequency.value = 120; low.gain.value = 2.5; const high = ctx.createBiquadFilter(); high.type = "highshelf"; high.frequency.value = 8500; high.gain.value = 3.2; /* 4️⃣ Stereo widen (space spirit) */ const merger = ctx.createChannelMerger(2); const splitter = ctx.createChannelSplitter(2); const widen = ctx.createGain(); widen.gain.value = 0.12; /* 5️⃣ Reverb (Earth cave spirit) */ const reverb = ctx.createConvolver(); // Simple impulse tail const ir = ctx.createBuffer(2, 44100, 44100); for (let ch = 0; ch < 2; ch++) { const data = ir.getChannelData(ch); for (let i = 0; i < data.length; i++) { data[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / data.length, 2); } } reverb.buffer = ir; /* 6️⃣ Output limiter (clean loud) */ const limiter = ctx.createDynamicsCompressor(); limiter.threshold.value = -3; limiter.knee.value = 0; limiter.ratio.value = 20; limiter.attack.value = 0.005; limiter.release.value = 0.08; /* Chain */ source.connect(saturator) .connect(comp) .connect(low) .connect(high) .connect(splitter); splitter.connect(merger, 0, 0); splitter.connect(widen); widen.connect(merger, 0, 1); merger.connect(reverb);
